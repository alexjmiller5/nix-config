# Manual Steps Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the hand-written `MANUAL-<host>.md` files with steps declared next to the nix module that needs them, generated (committed) manuals with a provenance header, a drift check, and an installed `manual-check` verifier.

**Architecture:** A shared option module `modules/manual-steps.nix` defines `manual.steps`; a darwin-only module `modules/manual.nix` merges darwin + Home Manager steps, renders `MANUAL-<host>.md` into `system.build.manual`, and installs `manual-check`. The flake exposes the renders as packages and a freshness check; the justfile's `manual` recipe writes them into the repo.

**Tech Stack:** nix-darwin + home-manager module system, `pkgs.writeText` / `writeShellApplication`, `just`, python3 tests (repo convention: `nix eval --impure --json` fixtures).

**Spec:** `docs/superpowers/specs/2026-09-15-manual-steps-design.md`

## Global Constraints

- Phases are exactly `bootstrap`, `after-switch`, `snapshot`; ordering is phase, then `owner`, then id.
- The generated header line is `<!-- GENERATED by \`just manual\` from https://github.com/alexjmiller5/nix-config/commit/<rev> - edit the declaring nix module or docs/manual/<host>-bootstrap.md, then rerun. -->`; freshness compares everything AFTER the first line.
- `manual-check` verifies only; it never performs a step. No `timeout` (macOS lacks it).
- Any exported homeModule that declares steps must itself `imports = [ ../modules/manual-steps.nix ]` so external consumers still evaluate.
- Nothing personal in step bodies beyond what the current manuals already carry (public repo).
- `nix fmt` before every commit; `just check` must stay green.

  lib = pkgs.lib;
  render = import (ROOT + "/modules/manual-render.nix") { inherit lib; };
  steps = {

             body = "do zeta"; verify = "true"; redo = "after z"; };

              body = "do alpha"; verify = null; redo = null; };

in render { host = "fixture"; rev = "abc123"; bootstrap = "# Boot\nfirst\n"; docs = [ "docs/x.md" ]; inherit steps; }
'''.replace("ROOT", json.dumps(str(ROOT)))

`modules/manual-steps.nix`:
```nix
{ lib, ... }:
# Human steps nix cannot perform (TCC, Keychain ACLs, sign-ins, pairings).
# Declared next to the module that needs them; rendered into MANUAL-<host>.md
# and verified by `manual-check` (modules/manual.nix). Importable by both
# nix-darwin and Home Manager modules.
let
  step = lib.types.submodule {
    options = {
      title = lib.mkOption { type = lib.types.str; };
      owner = lib.mkOption {
        type = lib.types.str;
        description = "Module, package or app the step belongs to (groups the manual).";
      };
      phase = lib.mkOption {
        type = lib.types.enum [ "bootstrap" "after-switch" "snapshot" ];
        default = "after-switch";
      };
      desktop = lib.mkOption {
        type = lib.types.bool;
        default = false;
        description = "Must run from the logged-in desktop / Screen Sharing.";
      };
      body = lib.mkOption { type = lib.types.lines; };
      verify = lib.mkOption {
        type = lib.types.nullOr lib.types.str;
        default = null;
        description = "Shell that exits 0 once the step is done; null = unverifiable.";
      };
      redo = lib.mkOption {
        type = lib.types.nullOr lib.types.str;
        default = null;
        description = "When the step must be repeated.";
      };
    };
  };
in
{
  options.manual.steps = lib.mkOption {
    type = lib.types.attrsOf step;
    default = { };
  };
}
```

`modules/manual-render.nix` (pure function, unit-testable):
```nix
{ lib }:
{ host, rev, bootstrap, docs, steps }:
let
  phases = [ "after-switch" "snapshot" ];
  headings = { "after-switch" = "## After the first switch"; snapshot = "## Snapshots (the list IS the declaration)"; };
  ordered = phase:
    lib.sort (a: b: a.owner < b.owner || (a.owner == b.owner && a.id < b.id))
      (lib.mapAttrsToList (id: s: s // { inherit id; })
        (lib.filterAttrs (_: s: s.phase == phase) steps));
  renderStep = s: ''
    ### ${s.title} (${s.owner} · ${if s.desktop then "desktop" else "shell"})

    ${lib.removeSuffix "\n" s.body}
    ${lib.optionalString (s.verify != null) "\nVerify: `${s.verify}`"}${lib.optionalString (s.redo != null) "\nRedo: ${s.redo}"}
  '';
  section = phase: lib.optionalString (ordered phase != [ ]) ''
    ${headings.${phase}}

    ${lib.concatMapStringsSep "\n" renderStep (ordered phase)}
  '';
in
''
  <!-- GENERATED by `just manual` from https://github.com/alexjmiller5/nix-config/commit/${rev} - edit the declaring nix module or docs/manual/${host}-bootstrap.md, then rerun. -->
  # MANUAL-${host}.md - steps nix cannot do

  Rendered from `manual.steps` (modules/manual-steps.nix). Check what this
  machine still owes with `manual-check`.

  ${lib.removeSuffix "\n" bootstrap}

  ${lib.concatMapStringsSep "\n" section phases}## Reference

  ${lib.concatMapStringsSep "\n" (d: "- [${d}](${d})") docs}
''
```

`modules/manual.nix` (darwin):
```nix
{ config, lib, pkgs, inputs, username, ... }:
let
  render = import ./manual-render.nix { inherit lib; };
  hmSteps = config.home-manager.users.${username}.manual.steps or { };
  dup = lib.intersectLists (lib.attrNames config.manual.steps) (lib.attrNames hmSteps);
  steps = assert lib.assertMsg (dup == [ ]) "manual.steps declared in both darwin and home-manager: ${toString dup}";
    config.manual.steps // hmSteps;
  rev = inputs.self.rev or inputs.self.dirtyRev or "unknown";
  host = config.manual.host;
  text = render {
    inherit host rev steps;
    docs = config.manual.docs;
    bootstrap = builtins.readFile (../docs/manual + "/${host}-bootstrap.md");
  };
  withVerify = lib.filterAttrs (_: s: s.verify != null) steps;
  manualCheck = pkgs.writeShellApplication {
    name = "manual-check";
    text = ''
      todo=0
      report() { printf '%-6s %-22s %s\n' "$1" "$2" "$3"; }
      ${lib.concatStringsSep "\n" (lib.mapAttrsToList (id: s: ''
        if (${s.verify}) >/dev/null 2>&1; then report done ${lib.escapeShellArg s.owner} ${lib.escapeShellArg s.title}
        else report todo ${lib.escapeShellArg s.owner} ${lib.escapeShellArg s.title}; todo=1; fi'') withVerify)}
      ${lib.concatStringsSep "\n" (lib.mapAttrsToList (id: s:
        "report manual ${lib.escapeShellArg s.owner} ${lib.escapeShellArg s.title}")
        (lib.filterAttrs (_: s: s.verify == null) steps))}
      exit $todo
    '';
  };
in
{
  imports = [ ./manual-steps.nix ];
  options.manual = {
    host = lib.mkOption { type = lib.types.str; description = "Host slug: MANUAL-<host>.md and docs/manual/<host>-bootstrap.md"; };
    docs = lib.mkOption { type = lib.types.listOf lib.types.str; default = [ ]; description = "Repo-relative docs linked under Reference."; };
  };
  config = {
    home-manager.sharedModules = [ ./manual-steps.nix ];
    system.build.manual = pkgs.writeText "MANUAL-${host}.md" text;
    environment.systemPackages = [ manualCheck ];
  };
}
```

manual:
    #!/usr/bin/env bash

      fresh="$(nix build --no-link --print-out-paths .#manual-$h)"

    done
